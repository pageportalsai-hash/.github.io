function segmentHitsRec51(rec,ax,az,bx,bz,margin=.10){
  const a=localXZ50(rec,ax,az),b=localXZ50(rec,bx,bz);
  const minX=-rec.w/2-margin,maxX=rec.w/2+margin,minZ=-rec.d/2-margin,maxZ=rec.d/2+margin;
  let t0=0,t1=1;
  const clipAxis=(p0,d,min,max)=>{
    if(Math.abs(d)<1e-6)return p0>=min&&p0<=max;
    let u0=(min-p0)/d,u1=(max-p0)/d;if(u0>u1){const q=u0;u0=u1;u1=q;}
    t0=Math.max(t0,u0);t1=Math.min(t1,u1);return t0<=t1;
  };
  if(!clipAxis(a.x,b.x-a.x,minX,maxX))return false;
  if(!clipAxis(a.z,b.z-a.z,minZ,maxZ))return false;
  return t1>.035&&t0<.965;
}
function obstruction51(n,target){
  const ax=n.group.position.x,az=n.group.position.z,bx=target.x,bz=target.z;
  for(const rec of buildingNav50){
    if(!rec?.g?.parent||rec.g.visible===false)continue;
    if(insideBuilding50(rec,ax,az,.05)||insideBuilding50(rec,bx,bz,.05))continue;
    const mx=(ax+bx)*.5,mz=(az+bz)*.5,reach=Math.hypot(bx-ax,bz-az)*.5+Math.max(rec.w,rec.d);
    if(Math.hypot(rec.x-mx,rec.z-mz)>reach)continue;
    if(segmentHitsRec51(rec,ax,az,bx,bz,.12))return rec;
  }
  return null;
}
