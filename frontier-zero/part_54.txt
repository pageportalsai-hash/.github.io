function floorOpening54(state,floor){
  const {g,rec,floorH,stairX,holeW,holeD}=state;
  const y=floor*floorH-.07,iw=Math.max(4,rec.w-.72),id=Math.max(4,rec.d-.72);
  const minX=-iw/2,maxX=iw/2,minZ=-id/2,maxZ=id/2;
  const hMinX=Math.max(minX+.25,stairX-holeW/2),hMaxX=Math.min(maxX-.25,stairX+holeW/2);
  const hMinZ=Math.max(minZ+.25,-holeD/2),hMaxZ=Math.min(maxZ-.25,holeD/2);
  slab54(g,hMinX-minX,id,(minX+hMinX)/2,y,0);
  slab54(g,maxX-hMaxX,id,(hMaxX+maxX)/2,y,0);
  slab54(g,hMaxX-hMinX,hMinZ-minZ,(hMinX+hMaxX)/2,y,(minZ+hMinZ)/2);
  slab54(g,hMaxX-hMinX,maxZ-hMaxZ,(hMinX+hMaxX)/2,y,(hMaxZ+maxZ)/2);

  // Guard rail around the open stairwell; leave the flight ends open as landings.
  const railY=y+.58;
  mesh54(g,new THREE.BoxGeometry(.06,1.05,holeD-.15),railMat54,hMinX-.08,railY,0,false);
  mesh54(g,new THREE.BoxGeometry(holeW,.06,.06),railMat54,stairX,railY,hMinZ-.08,false);
  mesh54(g,new THREE.BoxGeometry(holeW,.06,.06),railMat54,stairX,railY,hMaxZ+.08,false);
  // Soft ceiling panels keep the interior bright without expensive point lights.

function flight54(state,f){
  const {g,floorH,stairX,run}=state,steps=15,width=1.42,dir=f%2===0?1:-1;
  // Two parallel lanes make the switchback unambiguous: one flight can descend
  // while the next flight rises from the same landing without occupying the same path.
  const lane=(f%2===0?-1:1)*.82,x=stairX+lane;
  const z0=dir>0?-run/2:run/2,z1=-z0,rise=floorH/steps;
  const flight={f,x,width,run,z0,z1,dir};
  for(let i=0;i<steps;i++){
    const t=(i+.5)/steps,z=z0+(z1-z0)*t,y=f*floorH+(i+1)*rise-.07;
    const step=mesh54(g,new THREE.BoxGeometry(width,.16,run/steps+.035),stairMat54,x,y,z,false);
    step.userData.stairVisual54=true;
  }
  // Rail only on the room-facing edge; the tower wall is the outside edge.
  const innerX=x+(lane<0?-width/2-.10:width/2+.10);
  mesh54(g,new THREE.BoxGeometry(.055,1.00,run+.20),railMat54,innerX,f*floorH+floorH*.53,0,false);
  // Wide landings bridge both parallel flights at each turn.
  mesh54(g,new THREE.BoxGeometry(state.holeW-.22,.14,1.08),trimMat54,stairX,f*floorH-.03,z0,false);
  mesh54(g,new THREE.BoxGeometry(state.holeW-.22,.14,1.08),trimMat54,stairX,(f+1)*floorH-.03,z1,false);
  return flight;
}

function stairCandidate54(state,local){
  const c=[];
  for(const fi of [state.floor,state.floor-1]){
    if(fi<0||fi>=state.flights.length)continue;
    const f=state.flights[fi];
    if(Math.abs(local.x-f.x)>f.width/2+.38||Math.abs(local.z)>f.run/2+.45)continue;

  }

}
