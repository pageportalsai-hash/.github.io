let activeEmote60=null;
let emoteStart60=0;
let emotePrevThird60=false;
const EMOTE_DURATION60=8200;

const emoteHud60=document.createElement('div');
emoteHud60.style.cssText='position:fixed;left:50%;bottom:54px;transform:translateX(-50%);z-index:72;display:none;padding:7px 12px;border-radius:10px;background:rgba(8,12,15,.76);border:1px solid rgba(255,255,255,.20);color:#fff;font:900 11px system-ui;letter-spacing:.06em;pointer-events:none;text-shadow:0 1px 2px #000';
emoteHud60.textContent='1 · GRIDDY';
document.body.appendChild(emoteHud60);

function resetEmotePose60(){
  if(typeof playerRig35==='undefined'||!playerRig35)return;
  const r=playerRig35;
  for(const p of [r.lArm,r.rArm,r.lFore,r.rFore,r.lLeg,r.rLeg,r.lShin,r.rShin,r.hips,r.torso,r.head]){
    if(p){p.rotation.set(0,0,0);}
  }
  if(r.hips)r.hips.position.y=.84;
  if(r.torso)r.torso.position.y=1.33;
}

function stopEmote60(silent=false){
  if(!activeEmote60)return;
  activeEmote60=null;
  emoteHud60.style.display='none';
  resetEmotePose60();
  if(typeof setViewMode16==='function'&&thirdPerson16!==emotePrevThird60)setViewMode16(emotePrevThird60);
  if(!silent)message('EMOTE STOPPED',1.0);
}

function startGriddy60(){
  if(activeCar19||player.dead||(typeof lobbyActive17!=='undefined'&&lobbyActive17)||(typeof matchWon24!=='undefined'&&matchWon24)){
    message('CAN’T EMOTE RIGHT NOW',1.2);
    return;
  }
  if(activeEmote60==='griddy'){stopEmote60();return;}
  stopEmote60(true);
  activeEmote60='griddy';emoteStart60=performance.now();
  emotePrevThird60=!!thirdPerson16;
  if(typeof setViewMode16==='function'&&!thirdPerson16)setViewMode16(true);
  firing=false;
  emoteHud60.style.display='block';
  message('GRIDDY EMOTE · PRESS 1 AGAIN TO STOP',1.8);
}

addEventListener('keydown',e=>{
  if((e.code==='Digit1'||e.code==='Numpad1')&&!e.repeat){
    const tag=e.target?.tagName;
    if(tag==='INPUT'||tag==='TEXTAREA'||tag==='SELECT')return;
    e.preventDefault();
    startGriddy60();
  }
});

const doActionBeforeEmote60=doAction;
doAction=function(){
  if(activeEmote60){firing=false;return;}
  return doActionBeforeEmote60();
};

const updatePlayerBeforeEmote60=updatePlayer;
updatePlayer=function(dt){
  const before=player.pos.clone();
  updatePlayerBeforeEmote60(dt);
  if(!activeEmote60)return;

  if(activeCar19||player.dead||(typeof lobbyActive17!=='undefined'&&lobbyActive17)){
    stopEmote60(true);return;
  }

  // Keep the player planted while emoting, but keep terrain/stair height valid.
  player.pos.x=before.x;player.pos.z=before.z;
  const tower=(typeof activeTower54!=='undefined')?activeTower54:null;
  if(!tower){
    const y=groundY(player.pos.x,player.pos.z)+player.height;
    const dy=y-player.pos.y;player.pos.y=y;camera.position.y+=dy;
  }
  firing=false;

  const elapsed=performance.now()-emoteStart60;
  if(elapsed>=EMOTE_DURATION60){stopEmote60(true);return;}

  if(typeof playerRig35==='undefined'||!playerRig35)return;
  const r=playerRig35,t=elapsed*.001;
  const beat=t*6.0;
  const step=Math.sin(beat),alt=Math.sin(beat+Math.PI);
  const bounce=Math.abs(Math.sin(beat))*0.055;
  const side=Math.sin(beat*.5)*0.055;

  // Griddy-style: alternating quick steps with loose arm swings and heel rhythm.
  if(r.hips){
    r.hips.position.y=.84+bounce;
    r.hips.rotation.y=Math.sin(beat*.5)*.10;
    r.hips.rotation.z=side*.35;
  }
  if(r.torso){
    r.torso.rotation.x=-.08+Math.abs(step)*.035;
    r.torso.rotation.y=Math.sin(beat*.5)*.12;
    r.torso.rotation.z=-side*.55;
  }
  if(r.head){
    r.head.rotation.x=.03+Math.sin(beat*.5)*.025;
    r.head.rotation.z=Math.sin(beat*.5)*.035;
  }

  if(r.lLeg){r.lLeg.rotation.x=step>.0?-.58:.24;r.lLeg.rotation.z=.055;}
  if(r.rLeg){r.rLeg.rotation.x=alt>.0?-.58:.24;r.rLeg.rotation.z=-.055;}
  if(r.lShin)r.lShin.rotation.x=step>.0?.72:.10;
  if(r.rShin)r.rShin.rotation.x=alt>.0?.72:.10;

  if(r.lArm){
    r.lArm.rotation.x=-.72+alt*.33;
    r.lArm.rotation.z=-.38-.12*step;
    r.lArm.rotation.y=-.10;
  }
  if(r.rArm){
    r.rArm.rotation.x=-.72+step*.33;
    r.rArm.rotation.z=.38+.12*alt;
    r.rArm.rotation.y=.10;
  }
  if(r.lFore)r.lFore.rotation.x=-.88+Math.max(0,step)*.22;
  if(r.rFore)r.rFore.rotation.x=-.88+Math.max(0,alt)*.22;

  // Keep the avatar centered and facing the player's current facing direction.
  if(typeof playerAvatar16!=='undefined'){
    playerAvatar16.position.x=player.pos.x;
    playerAvatar16.position.z=player.pos.z;
    playerAvatar16.rotation.y=player.yaw;
  }
};

window.__frontierZeroEmotes60={keys:{1:'Griddy'},durationMs:EMOTE_DURATION60};
