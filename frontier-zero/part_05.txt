// Replace automatic long-range detection with the requested close proximity trigger.
// Enemies automatically aggro when the player is within about 4 meters.
// Shooting an enemy still alerts that enemy, and walls/trees/terrain still block its shots.
updateNPCs=function(dt){
  for(const n of npcs){
    if(n.dead)continue;
    const p=n.group.position,to=player.pos.clone().sub(p);to.y=0;const dist=to.length();
    const proximityAggro=dist<=4.2;
    if(proximityAggro)n.alert=Math.max(n.alert,6);
    else n.alert=Math.max(0,n.alert-dt);
    const see=npcCanSee(n);
    let move=new THREE.Vector3();
    if(n.alert>0){
      const dir=to.lengthSq()>.0001?to.clone().normalize():new THREE.Vector3(0,0,-1);
      if(dist>5.2)move.copy(dir);
      else {move.set(-dir.z,0,dir.x).multiplyScalar(n.strafe*.6);if(dist<2.4)move.addScaledVector(dir,-.45);}
      n.shotCd-=dt;
      if(see&&dist<32&&n.shotCd<=0){
        n.shotCd=rand(.55,1.05);
        const accuracy=clamp(.54-dist/95,.28,.52); // close enemies are dangerous but still miss often
        if(Math.random()<accuracy)damagePlayer(rand(5,9));else tone(65,.025,.008,'square');
      }
    }else{
      n.repath-=dt;
      if(n.repath<=0||p.distanceTo(n.patrol)<4){n.repath=rand(4,8);n.patrol.set(p.x+rand(-24,24),0,p.z+rand(-24,24));}
      move.copy(n.patrol).sub(p);move.y=0;if(move.length()>1)move.normalize();
    }
    if(move.lengthSq()>0){
      const speed=n.alert>0?3.9:1.55,nx=p.x+move.x*speed*dt,nz=p.z+move.z*speed*dt;
      if(!blockedPosition(nx,nz)){p.x=nx;p.z=nz;}else{n.strafe*=-1;n.patrol.set(p.x+rand(-14,14),0,p.z+rand(-14,14));}
      p.y=groundY(p.x,p.z);n.group.rotation.y=Math.atan2(move.x,move.z)+Math.PI;
    }
  }
};
