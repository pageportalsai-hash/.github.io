function updateToolVisual15(){
  const w=currentGear();
  const isTool=w&&(w.type==='axe'||w.type==='pickaxe'||w.type==='dig');
  toolHolder15.visible=!!isTool;gunHolder.visible=!isTool;
  axeModel15.visible=w?.type==='axe';pickModel15.visible=w?.type==='pickaxe';digModel15.visible=w?.type==='dig';
  if(isTool){
    toolHolder15.rotation.set(.02,0,-.05);
    toolHolder15.position.set(.34,-.34,-.70);
  }
}
