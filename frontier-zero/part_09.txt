// Player travel across full map while retaining existing movement behavior.
updatePlayer=function(dt){
  baseUpdatePlayerCompact(dt);
  // Desktop keyboard input is added to the same player movement system below.
  const kf=new THREE.Vector3(-Math.sin(player.yaw),0,-Math.cos(player.yaw));
  const kr=new THREE.Vector3(Math.cos(player.yaw),0,-Math.sin(player.yaw));
  const km=new THREE.Vector3();
  if(keys.KeyW||keys.ArrowUp)km.add(kf);
  if(keys.KeyS||keys.ArrowDown)km.sub(kf);
  if(keys.KeyD||keys.ArrowRight)km.add(kr);
  if(keys.KeyA||keys.ArrowLeft)km.sub(kr);
  if(km.lengthSq()>1)km.normalize();
  if(km.lengthSq()>.001){const sprint=(keys.ShiftLeft||keys.ShiftRight)?1.55:1;const nx=player.pos.x+km.x*player.speed*sprint*dt,nz=player.pos.z+km.z*player.speed*sprint*dt;if(!blockedPosition(nx,nz)){player.pos.x=nx;player.pos.z=nz;}}
  player.pos.x=clamp(player.pos.x,-994,994);player.pos.z=clamp(player.pos.z,-994,994);
  player.pos.y=groundY(player.pos.x,player.pos.z)+player.height;camera.position.copy(player.pos);
};
