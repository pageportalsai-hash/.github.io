// High-resolution 500m terrain layer with procedural micro-texture.
function makeGroundTexture(){
  const c=document.createElement('canvas');c.width=c.height=256;const ctx=c.getContext('2d');
  ctx.fillStyle='#a4a59a';ctx.fillRect(0,0,256,256);
  for(let i=0;i<7000;i++){
    const v=(105+Math.random()*75)|0;ctx.fillStyle=`rgba(${v},${v},${v},${rand(.025,.11)})`;
    const s=Math.random()<.95?1:2;ctx.fillRect(Math.random()*256,Math.random()*256,s,s);
  }
  for(let i=0;i<55;i++){
    ctx.strokeStyle=`rgba(70,65,54,${rand(.025,.07)})`;ctx.lineWidth=rand(.3,1.1);ctx.beginPath();
    const x=rand(0,256),y=rand(0,256);ctx.moveTo(x,y);ctx.lineTo(x+rand(-18,18),y+rand(-18,18));ctx.stroke();
  }
  const t=new THREE.CanvasTexture(c);t.wrapS=t.wrapT=THREE.RepeatWrapping;t.repeat.set(36,36);t.colorSpace=THREE.SRGBColorSpace;return t;
}
