// ---------- Loot crates disappear completely after opening ----------
function removeLootCrate17(c){
  if(!c?.group)return;
  c.npcReserved14=null;
  c.group.visible=false;
  scene.remove(c.group);
}

const lootCrateBeforeDisappear17=lootCrate;
lootCrate=function(c){
  if(!c||c.looted)return;
  lootCrateBeforeDisappear17(c);
  if(c.looted)removeLootCrate17(c);
};

const npcOpenLootBeforeDisappear17=npcOpenLoot14;
npcOpenLoot14=function(n,c){
  const opened=npcOpenLootBeforeDisappear17(n,c);
  if(opened)removeLootCrate17(c);
  return opened;
};

// ---------- Spawn safety: never spawn in the ocean/water or marsh pools ----------
function spawnIsDry17(x,z){
  // Main world water plane sits at y=-4.7. Keep a healthy safety margin above it.
  if(groundY(x,z)<=water.position.y+1.25)return false;
  // Marsh biome contains raised water pools; don't use it for player spawn points.
  if(typeof biomeAt==='function'&&typeof BIOMES!=='undefined'&&biomeAt(x,z)===BIOMES.marsh)return false;
  return true;
}

randomSafeSpawn=function(){
  for(let tries=0;tries<220;tries++){
    const c=populationCenters[(Math.random()*populationCenters.length)|0];
    let x,z;
    if(Math.random()<.62){
      const a=rand(0,Math.PI*2),r=rand(70,135);
      x=c[0]+Math.cos(a)*r;z=c[1]+Math.sin(a)*r;
    }else{
      x=rand(-455,455);z=rand(-455,455);
    }
    if(Math.abs(x)>470||Math.abs(z)>470)continue;
    if(!spawnIsDry17(x,z))continue;
    if(blockedPosition(x,z))continue;
    let tooClose=false;
    for(const n of npcs){
      if(!n.dead&&Math.hypot(n.group.position.x-x,n.group.position.z-z)<16){tooClose=true;break;}
    }
    if(tooClose)continue;
    return {x,z};
  }
  // Dry central fallback if random sampling somehow fails.
  const fallbacks=[[0,80],[120,80],[-120,80],[0,180],[180,0],[-180,0]];
  for(const [x,z] of fallbacks)if(spawnIsDry17(x,z)&&!blockedPosition(x,z))return{x,z};
  return{x:0,z:0};
};

function finishLobby17(){
  if(!lobbyActive17)return;
  lobbyActive17=false;
  clearInterval(lobbyTimer17);

  // Start with exactly the selected Common gun plus the three harvesting tools.
  const chosen=gear.find(w=>w.type==='gun'&&w.name===selectedCommon17);
  const tools=gear.filter(w=>w.type!=='gun');
  if(chosen){
    chosen.rarity='Common';
    chosen.mag=chosen.magSize;
    // Keep a sensible full starting reserve.
    if(selectedCommon17==='Rifle')chosen.reserve=Math.max(chosen.reserve,120);
    else if(selectedCommon17==='Pistol')chosen.reserve=Math.max(chosen.reserve,60);
    else if(selectedCommon17==='Shotgun')chosen.reserve=Math.max(chosen.reserve,30);
    gear.splice(0,gear.length,chosen,...tools);
    gearIndex=0;
  }

  // Enter the match at a new dry, safe random location.
  applyRandomSpawn();
  player.health=100;player.dead=false;firing=false;
  lobby17.remove();
  updateHUD();
  if(typeof setViewMode16==='function'&&thirdPerson16)setViewMode16(true);
  message(`MATCH STARTED · Common ${selectedCommon17} selected`,2.8);
}

const lobbyTimer17=setInterval(()=>{
  lobbySeconds17--;
  lobbyCountdown17.textContent=lobbySeconds17>0?`Starting in ${lobbySeconds17} seconds`:'Starting match…';
  if(lobbySeconds17<=0)finishLobby17();
},1000);
