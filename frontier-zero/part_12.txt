// Remove the former 1500m hard boundary.
const old1500Edges=[];
for(const o of [...blockers,...playerStructures]){
  if(!o?.parent)continue;
  const p=new THREE.Vector3();o.getWorldPosition(p);
  const b=new THREE.Box3().setFromObject(o),s=new THREE.Vector3();b.getSize(s);
  if((Math.abs(Math.abs(p.x)-748)<7||Math.abs(Math.abs(p.z)-748)<7)&&Math.max(s.x,s.z)>1300)old1500Edges.push(o);
}
for(const o of old1500Edges){
  scene.remove(o);
  for(let i=blockers.length-1;i>=0;i--)if(blockers[i]===o)blockers.splice(i,1);
  for(let i=playerStructures.length-1;i>=0;i--)if(playerStructures[i]===o)playerStructures.splice(i,1);
}

// New 1000m x 1000m physical boundary.
const edge1000Mat=new THREE.MeshStandardMaterial({color:0x414742,roughness:1});
function edge1000(x,z,sx,sz){
  const m=new THREE.Mesh(new THREE.BoxGeometry(sx,8,sz),edge1000Mat);
  m.position.set(x,groundY(x,z)+2,z);m.castShadow=m.receiveShadow=true;scene.add(m);blockers.push(m);playerStructures.push(m);
}
edge1000(0,-498,1000,4);edge1000(0,498,1000,4);edge1000(-498,0,4,1000);edge1000(498,0,4,1000);

    if(removed.has(owner))npcHitMeshes.splice(i,1);

  npcs.splice(50);
}

// Construction obeys the smaller map.

  if(preview&&(Math.abs(preview.position.x)>490||Math.abs(preview.position.z)>490)){message('Map edge — build inside the 1000m zone');return;}
  buildBefore1000();
};

// New random safe spawn logic for the 1000m map.
randomSafeSpawn=function(){
  for(let tries=0;tries<120;tries++){
    const c=populationCenters[(Math.random()*populationCenters.length)|0];
    let x,z;
    if(Math.random()<.6){const a=rand(0,Math.PI*2),r=rand(65,125);x=c[0]+Math.cos(a)*r;z=c[1]+Math.sin(a)*r;}
    else{x=rand(-455,455);z=rand(-455,455);}
    if(Math.abs(x)>470||Math.abs(z)>470||blockedPosition(x,z))continue;
    let close=false;for(const n of npcs){if(!n.dead&&Math.hypot(n.group.position.x-x,n.group.position.z-z)<12){close=true;break;}}
    if(!close)return{x,z};
  }
  return{x:rand(-180,180),z:rand(-180,180)};
};
applyRandomSpawn();

const ownedGunKeys=new Set();
for(const w of gear)if(w.type==='gun')ownedGunKeys.add((w.rarity||'Common')+'|'+w.name);

function rollRarity(){
  const entries=Object.entries(RARITY),total=entries.reduce((s,[,v])=>s+v.weight,0);let r=Math.random()*total;
  for(const [name,v] of entries){r-=v.weight;if(r<=0)return name;}return 'Common';
}
function grantRandomGun(){
  const rarity=rollRarity();let candidates=gunCatalog.filter(g=>g.rarity===rarity&&!ownedGunKeys.has(g.rarity+'|'+g.name));
  if(!candidates.length)candidates=gunCatalog.filter(g=>!ownedGunKeys.has(g.rarity+'|'+g.name));
  if(!candidates.length)return null;
  const src=candidates[(Math.random()*candidates.length)|0];
  const w={name:src.name,type:'gun',rarity:src.rarity,magSize:src.magSize,mag:src.magSize,reserve:src.reserve,damage:src.damage,rate:src.rate,spread:src.spread,pellets:src.pellets};
  gear.push(w);ownedGunKeys.add(src.rarity+'|'+src.name);return w;
}
