// ---------- Replace floating roads with roads that conform to terrain ----------
// Hide every old flat asphalt/dirt road mesh from earlier updates.
scene.traverse(o=>{
  if(!o?.isMesh)return;
  if((typeof asphaltMat!=='undefined'&&o.material===asphaltMat)||(typeof dirtRoadMat!=='undefined'&&o.material===dirtRoadMat))o.visible=false;
});

const roadGroundMat=new THREE.MeshStandardMaterial({color:0x343638,roughness:.93,metalness:.015});
const roadDirtGroundMat=new THREE.MeshStandardMaterial({color:0x705c45,roughness:1,metalness:0});

function terrainRoad(ax,az,bx,bz,width=7,material=roadGroundMat){
  const dx=bx-ax,dz=bz-az,len=Math.hypot(dx,dz),steps=Math.max(6,Math.ceil(len/9));
  const nx=-dz/len*(width/2),nz=dx/len*(width/2),verts=[],inds=[],uv=[];
  for(let i=0;i<=steps;i++){
    const t=i/steps,x=ax+dx*t,z=az+dz*t;
    const x1=x+nx,z1=z+nz,x2=x-nx,z2=z-nz;
    verts.push(x1,groundY(x1,z1)+.055,z1,x2,groundY(x2,z2)+.055,z2);
    uv.push(0,t*len/12,1,t*len/12);
    if(i<steps){const a=i*2,b=a+1,c=a+2,d=a+3;inds.push(a,c,b,c,d,b);}
  }
  const g=new THREE.BufferGeometry();g.setAttribute('position',new THREE.Float32BufferAttribute(verts,3));g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));g.setIndex(inds);g.computeVertexNormals();
  const m=new THREE.Mesh(g,material);m.receiveShadow=true;scene.add(m);return m;
}

function settlementColors(x,z){
  const b=biomeAt(x,z);
  if(b===BIOMES.badlands)return [0x826955,0x4b4037];
  if(b===BIOMES.alpine)return [0x737a7b,0x414749];
  if(b===BIOMES.marsh)return [0x5a655f,0x394443];
  if(b===BIOMES.pine)return [0x626a61,0x3b423d];
  return [0x756f65,0x414546];
}

for(const s of settlementLayout){
  const count=s.type==='city'?9:5,[wall,roof]=settlementColors(s.x,s.z);
  for(let i=0;i<count;i++){
    const rows=s.type==='city'?3:2,cols=Math.ceil(count/rows),row=Math.floor(i/cols),col=i%cols;
    const spacing=s.type==='city'?18:20;
    const ox=(col-(cols-1)/2)*spacing+rand(-2,2),oz=(row-(rows-1)/2)*spacing+rand(-2,2);
    localBuilding(s.x+ox,s.z+oz,rand(7.2,11.5),rand(6.3,9.5),wall,roof,s.name+' Loot');
  }
  // small outpost on city/town edge adds more variation and loot.
  if(s.type==='city')makeOutpost(s.x+45,s.z-38,biomeAt(s.x,s.z)===BIOMES.badlands?'badlands':biomeAt(s.x,s.z)===BIOMES.alpine?'alpine':biomeAt(s.x,s.z)===BIOMES.marsh?'marsh':'pine');

// ---------- Random safe player spawning ----------
function randomSafeSpawn(){
  // Prefer a mixture of wilderness and settlement outskirts, never the exact same point.
  for(let tries=0;tries<120;tries++){
    let x,z;
    if(Math.random()<.62){
      const s=settlementLayout[(Math.random()*settlementLayout.length)|0],a=rand(0,Math.PI*2),r=rand(45,95);
      x=s.x+Math.cos(a)*r;z=s.z+Math.sin(a)*r;
    }else{x=rand(-690,690);z=rand(-690,690);}
    if(Math.abs(x)>710||Math.abs(z)>710)continue;
    if(blockedPosition(x,z))continue;
    let tooClose=false;
    for(const n of npcs){if(!n.dead&&Math.hypot(n.group.position.x-x,n.group.position.z-z)<10){tooClose=true;break;}}
    if(tooClose)continue;
    return {x,z};
  }
  return {x:rand(-300,300),z:rand(-300,300)};
}
function applyRandomSpawn(){
  const s=randomSafeSpawn();player.pos.x=s.x;player.pos.z=s.z;jumpOffset=0;jumpVelocity=0;onGround=true;
  player.pos.y=groundY(s.x,s.z)+player.height;camera.position.copy(player.pos);
}

// Randomize the initial spawn as soon as this upgrade loads.
applyRandomSpawn();

// Override death/respawn so each remaining life can respawn somewhere different.
playerDeath=function(){
  if(player.dead)return;player.dead=true;firing=false;player.lives=Math.max(0,(player.lives??5)-1);updateLivesHUD();
  if(player.lives<=0){message('No lives remaining',2);setTimeout(()=>{if(gameOverOverlay)gameOverOverlay.style.display='flex';},650);return;}
  message(`You died · ${player.lives} lives left · relocating…`,1.5);
  setTimeout(()=>{player.health=100;player.dead=false;applyRandomSpawn();updateHUD();message(`Respawned somewhere new · ${player.lives} lives left`,1.8);},1700);
};

// Keep every NPC inside the current 1500m map after the population increase.
const npcBefore100=updateNPCs;
updateNPCs=function(dt){
  npcBefore100(dt);
  for(const n of npcs){if(n.dead)continue;n.group.position.x=clamp(n.group.position.x,-720,720);n.group.position.z=clamp(n.group.position.z,-720,720);n.group.position.y=groundY(n.group.position.x,n.group.position.z);n.patrol.x=clamp(n.patrol.x,-710,710);n.patrol.z=clamp(n.patrol.z,-710,710);}
};
