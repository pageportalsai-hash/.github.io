  if(typeof lobbyActive17!=='undefined'&&lobbyActive17){
    if(typeof updatePlayerBeforeLobby17==='function')updatePlayerBeforeLobby17(dt);
    player.pos.x=clamp(player.pos.x,-6.65,6.65);player.pos.z=clamp(player.pos.z,-6.65,6.65);player.pos.y=skyLobbyY18+player.height+.32;
    camera.position.copy(player.pos);
    camera.rotation.set(player.pitch,player.yaw,0);

      playerAvatar16.position.set(player.pos.x,skyLobbyY18+.28,player.pos.z);playerAvatar16.rotation.y=player.yaw;

finishLobby17=function(){
  if(!lobbyActive17)return;
  const fallback=commonLoadoutDefs18.map(d=>d.name).filter(n=>!selectedLoadout18.includes(n));
  while(selectedLoadout18.length<2&&fallback.length)selectedLoadout18.push(fallback.shift());
  lobbyActive17=false;clearInterval(lobbyTimer17);

  const tools=gear.filter(w=>w.type!=='gun');
  const guns=selectedLoadout18.slice(0,2).map(name=>makeCommonLoadoutGun18(commonLoadoutDefs18.find(d=>d.name===name)));
  gear.splice(0,gear.length,...guns,...tools);gearIndex=0;

  const spawn=randomSafeSpawn();
  player.pos.set(spawn.x,groundY(spawn.x,spawn.z)+player.height,spawn.z);camera.position.copy(player.pos);
  player.health=100;player.dead=false;firing=false;
  if(typeof playerAvatar16!=='undefined')playerAvatar16.position.set(spawn.x,groundY(spawn.x,spawn.z),spawn.z);

  skyLobby18.visible=false;scene.remove(skyLobby18);
  if(typeof lobby17!=='undefined'&&lobby17.isConnected)lobby17.remove();
  updateHUD();
  message(`DEPLOYED · ${guns[0].name} + ${guns[1].name}`,3);
};
