// ============================================================================
// FRONTIER ZERO FLOATING SKY LOBBY UPDATE
// Player waits inside a floating sky box during the 30-second lobby, then
// teleports to a random safe dry location when the countdown ends.
// ============================================================================

const skyLobby18=new THREE.Group();scene.add(skyLobby18);
const skyLobbyY18=120;
const skyFloorMat18=new THREE.MeshStandardMaterial({color:0x56616a,roughness:.82,metalness:.12});
const skyWallMat18=new THREE.MeshStandardMaterial({color:0x7b8b96,roughness:.72,metalness:.08,transparent:true,opacity:.86});
const skyTrimMat18=new THREE.MeshStandardMaterial({color:0xd5b84a,roughness:.48,metalness:.32});
function skyBoxPart18(sx,sy,sz,x,y,z,mat){const m=new THREE.Mesh(new THREE.BoxGeometry(sx,sy,sz),mat);m.position.set(x,y,z);m.castShadow=true;m.receiveShadow=true;skyLobby18.add(m);return m;}
// 14m square floating room with waist-to-head height walls and open ceiling.
skyBoxPart18(14,.5,14,0,skyLobbyY18,0,skyFloorMat18);
skyBoxPart18(14,4,.35,0,skyLobbyY18+2,-6.8,skyWallMat18);
skyBoxPart18(14,4,.35,0,skyLobbyY18+2,6.8,skyWallMat18);
skyBoxPart18(.35,4,14,-6.8,skyLobbyY18+2,0,skyWallMat18);
skyBoxPart18(.35,4,14,6.8,skyLobbyY18+2,0,skyWallMat18);
// bright corner trims make it feel like a proper pre-match platform.
for(const [x,z] of [[-6.55,-6.55],[6.55,-6.55],[-6.55,6.55],[6.55,6.55]])skyBoxPart18(.18,4.4,.18,x,skyLobbyY18+2.15,z,skyTrimMat18);

function putPlayerInSkyLobby18(){
  player.pos.set(0,skyLobbyY18+player.height+.32,0);
  player.yaw=0;player.pitch=-.08;
  camera.position.copy(player.pos);
  if(typeof playerAvatar16!=='undefined'){
    playerAvatar16.position.set(0,skyLobbyY18+.28,0);
    playerAvatar16.rotation.y=0;
  }
}
putPlayerInSkyLobby18();

// While the lobby countdown is active, hold the player inside the floating box.
const updatePlayerBeforeSkyLobby18=updatePlayer;
updatePlayer=function(dt){
  if(typeof lobbyActive17!=='undefined'&&lobbyActive17){
    player.pos.x=clamp(player.pos.x,-5.9,5.9);
    player.pos.z=clamp(player.pos.z,-5.9,5.9);
    player.pos.y=skyLobbyY18+player.height+.32;
    camera.position.copy(player.pos);
    if(typeof playerAvatar16!=='undefined'){
      playerAvatar16.position.set(player.pos.x,skyLobbyY18+.28,player.pos.z);
      playerAvatar16.rotation.y=player.yaw;
    }
    return;
  }
  updatePlayerBeforeSkyLobby18(dt);
};

// Override the lobby finish so the player leaves the sky room and drops into
// a fresh random dry/safe location at exactly the end of the countdown.
const finishLobbyBeforeSky18=finishLobby17;
finishLobby17=function(){
  if(!lobbyActive17)return;
  // Let the existing function handle gun selection and general match setup.
  finishLobbyBeforeSky18();
  // Ensure the final position is randomized again after the lobby is removed.
  const spawn=randomSafeSpawn();
  player.pos.set(spawn.x,groundY(spawn.x,spawn.z)+player.height,spawn.z);
  camera.position.copy(player.pos);
  if(typeof playerAvatar16!=='undefined')playerAvatar16.position.set(spawn.x,groundY(spawn.x,spawn.z),spawn.z);
  skyLobby18.visible=false;
  scene.remove(skyLobby18);
  message('DROPPED INTO THE MAP · random safe spawn',2.6);
};

// Make lobby text explain that the player is waiting in the sky box.
if(typeof lobbyCountdown17!=='undefined'){
  const skyHint18=document.createElement('div');
  skyHint18.style.cssText='margin-top:10px;font-size:12px;opacity:.68';
  skyHint18.textContent='You are waiting inside the floating sky lobby. When the timer reaches 0 you will spawn randomly into the map.';
  lobbyCountdown17.insertAdjacentElement('afterend',skyHint18);
}

message('Floating sky lobby ready · random spawn when countdown ends',3.2);
