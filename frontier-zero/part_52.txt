function towerWallHeight52(rec){
  let h=0;
  for(const o of rec.g.children){
    if(!o?.isMesh||o.geometry?.type!=='BoxGeometry')continue;
    const p=o.geometry.parameters||{};
    if((p.height||0)>3.2&&((p.width||0)<1||(p.depth||0)<1))h=Math.max(h,p.height||0);
  }
  return h||12;
}
function towerMesh52(parent,geo,mat,x,y,z,shadow=true){
  const m=new THREE.Mesh(geo,mat);m.position.set(x||0,y||0,z||0);m.castShadow=shadow;m.receiveShadow=true;parent.add(m);return m;
}
function rebuildTowerPortals52(rec){
  rec.center=worldXZ50(rec,0,rec.d/2+.22);

  rec.inside=worldXZ50(rec,0,rec.d/2-1.35);
}
function makeFloorWithStairHall52(state,floor){
  const {group,rec,floorH}=state,w=rec.w,d=rec.d,y=floor*floorH-.08;
  const hallW=Math.min(4.6,Math.max(3.7,w*.36)),edgeD=1.45,sideW=Math.max(.65,(w-hallW)/2);
  const back=towerMesh52(group,new THREE.BoxGeometry(w-.5,.16,edgeD),towerFloorMat52,0,y,-d/2+edgeD/2+.25);
  const front=towerMesh52(group,new THREE.BoxGeometry(w-.5,.16,edgeD),towerFloorMat52,0,y,d/2-edgeD/2-.25);
  const midD=Math.max(1.6,d-edgeD*2-.5);
  const left=towerMesh52(group,new THREE.BoxGeometry(sideW,.16,midD),towerFloorMat52,-hallW/2-sideW/2,y,0);
  const right=towerMesh52(group,new THREE.BoxGeometry(sideW,.16,midD),towerFloorMat52,hallW/2+sideW/2,y,0);
  for(const m of [back,front,left,right]){m.userData.towerFloor52=floor;blockers.push(m);}
}
function makeFlight52(state,f){
  const {group,rec,floorH}=state,steps=13,run=Math.min(rec.d*.60,7.2),side=1.08*(f%2===0?-1:1),dir=f%2===0?1:-1;
  const z0=dir>0?-run/2:run/2,z1=-z0,rise=floorH/steps;
  const flight={f,side,dir,z0,z1,run,width:1.72};
  for(let s=0;s<steps;s++){
    const t=(s+.5)/steps,z=z0+(z1-z0)*t,y=f*floorH+(s+1)*rise-.06;
    const step=towerMesh52(group,new THREE.BoxGeometry(flight.width,.18,run/steps+.035),towerStairMat52,side,y,z);
    step.userData.towerStair52=true;blockers.push(step);
  }
  // Slim railings make the stair route visually obvious without blocking movement.
  for(const rx of [side-flight.width/2-.08,side+flight.width/2+.08]){
    const rail=towerMesh52(group,new THREE.BoxGeometry(.055,1.0,run+.2),towerRailMat52,rx,f*floorH+floorH*.52,0,false);
    rail.userData.decorative52=true;
  }
  return flight;
}

function towerAt52(x,z,margin=.16){

}
function flightProgress52(f,z){return clamp((z-f.z0)/(f.z1-f.z0),0,1);}
function stairCandidate52(state,local){
  const floor=state.floor,cands=[];
  for(const fi of [floor,floor-1]){
    if(fi<0||fi>=state.flights.length)continue;const f=state.flights[fi];
    if(Math.abs(local.x-f.side)>f.width/2+.34||Math.abs(local.z)>f.run/2+.38)continue;
    const t=flightProgress52(f,local.z),e=fi*state.floorH+t*state.floorH;
    cands.push({f,fi,t,e,d:Math.abs(e-state.elev)});
  }
  cands.sort((a,b)=>a.d-b.d);return cands[0]||null;
}

let activeTower52=null;
const updatePlayerBeforeFloors52=updatePlayer;
updatePlayer=function(dt){
  const prevX=player.pos.x,prevZ=player.pos.z,prevTower=activeTower52;
  const r=updatePlayerBeforeFloors52(dt);
  if(activeCar19){activeTower52=null;return r;}
  let state=towerAt52(player.pos.x,player.pos.z,.10);
  if(!state&&prevTower&&prevTower.floor>0){
    // Upper floors cannot walk out through the ground-floor doorway/windows.
    player.pos.x=prevX;player.pos.z=prevZ;state=prevTower;
  }
  if(!state){activeTower52=null;return r;}
  if(activeTower52!==state){state.floor=0;state.elev=0;activeTower52=state;}

  const local=localXZ50(state.rec,player.pos.x,player.pos.z),cand=stairCandidate52(state,local);
  if(cand){
    state.elev=cand.e;
    if(cand.fi===state.floor&&cand.t>.94&&state.floor<state.floors-1){state.floor++;state.elev=state.floor*state.floorH;}
    else if(cand.fi===state.floor-1&&cand.t<.06&&state.floor>0){state.floor--;state.elev=state.floor*state.floorH;}
  }else state.elev=state.floor*state.floorH;

  const oldY=player.pos.y,desiredY=state.baseY+state.elev+player.height,dy=desiredY-oldY;
  player.pos.y=desiredY;camera.position.y+=dy;
  if(typeof playerAvatar16!=='undefined')playerAvatar16.position.y+=dy;

    if(typeof lootBtn!=='undefined')lootBtn.style.display=nearbyLoot?'flex':'none';
  }
  return r;
};

// Move each tower's existing loot across multiple floors instead of stacking it
// all on the ground floor.
for(const state of towerStates52){
  const rec=state.rec;
  const crates=lootCrates.filter(c=>!c.looted&&c.group?.parent&&typeof c.label==='string'&&c.label.startsWith(rec.label51+' Loot'));
  for(let i=0;i<crates.length;i++){
    const c=crates[i],floor=i%state.floors,slot=(i*3+floor)%4;
    const lx=slot%2===0?-rec.w*.27:rec.w*.27,lz=slot<2?-rec.d*.27:rec.d*.27,p=worldXZ50(rec,lx,lz);
    c.group.position.set(p.x,state.baseY+floor*state.floorH+.42,p.z);
    c.towerFloor52=floor;c.towerState52=state;
  }
}

// Height-aware player loot proximity: upper-floor crates cannot be opened through
// the ceiling while standing underneath them.
findNearbyLoot=function(){
  let best=null,bestD=4.25;
  for(const c of lootCrates){
    if(!c||c.looted||!c.group?.parent)continue;
    const dx=c.group.position.x-player.pos.x,dz=c.group.position.z-player.pos.z,dy=c.group.position.y-(player.pos.y-player.height*.45),d=Math.hypot(dx,dz,dy);
    if(d<bestD){bestD=d;best=c;}
  }
  return best;
};

    for(const c of lootCrates){
      if(!c||c.looted||!c.group?.parent||(c.towerFloor52||0)>0)continue;
      if(c.supply47||c.npcDrop47)continue;

      if(d<bestD){best=c;bestD=d;}
    }
    return best;
  };
}
