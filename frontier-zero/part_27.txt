function weaponFamily27(w){
  const n=(w?.name||'').toLowerCase();
  if(n.includes('revolver')||n.includes('handcannon')||n.includes('cannon'))return 'revolver';
  if(n.includes('pistol'))return 'pistol';
  if(n.includes('shotgun'))return 'shotgun';
  if(n.includes('smg')||n.includes('compact auto')||n.includes('machine'))return 'smg';
  if(n.includes('dmr')||n.includes('marksman'))return 'dmr';
  return 'rifle';
}
function skinColors27(w){
  const s=(w?.skin?.name||w?.skin||'').toString().toLowerCase();
  if(s.includes('forest'))return [0x314637,0x56634e];
  if(s.includes('desert'))return [0x6c5840,0x9b7d54];
  if(s.includes('arctic'))return [0xb9c6ca,0x5e707b];
  if(s.includes('crimson'))return [0x552427,0x9d3d40];
  if(s.includes('midnight'))return [0x121720,0x394a72];
  if(s.includes('gold'))return [0x3b3427,0xc49a3d];
  if(s.includes('void'))return [0x21172a,0x7650b2];
  return [0x20262b,0x626b70];
}
let lastWeapon27='';
function applyRealWeapon27(){
  const w=currentGear();
  const isGun=!!w&&w.type==='gun';weaponModels27.visible=isGun&&!activeCar19;
  if(!isGun){for(const k in models27)models27[k].visible=false;return;}
  const family=weaponFamily27(w),key=family+'|'+w.name+'|'+(w.skin?.name||w.skin||'');
  for(const k in models27)models27[k].visible=k===family;
  if(key===lastWeapon27)return;lastWeapon27=key;
  const model=models27[family],cols=skinColors27(w);
  model.traverse(m=>{
    if(!m.isMesh||!m.material?.color)return;
    const metal=m.material.metalness??0;
    if(metal>.5){m.material=m.material.clone();m.material.color.setHex(cols[0]);}
    else if(m.material===gunPoly27||metal>.12){m.material=m.material.clone();m.material.color.setHex(cols[1]);}
  });
  if(model.userData.scope)model.userData.scope.visible=(family==='dmr'||['Rare','Epic','Legendary','Mythic','Secret'].includes(w.rarity));
}
