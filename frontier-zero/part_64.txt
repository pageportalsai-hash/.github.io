function candidateCover64(n,maxDist=40){
  const p=n.group.position;
  let best=null,bestScore=Infinity;
  const consider=(cx,cz,radius=2.2)=>{
    const nd=Math.hypot(cx-p.x,cz-p.z);if(nd>maxDist||nd<2.0)return;
    const vx=cx-player.pos.x,vz=cz-player.pos.z,vl=Math.hypot(vx,vz);if(vl<.001)return;
    const ux=vx/vl,uz=vz/vl;
    const x=cx+ux*(radius+1.0),z=cz+uz*(radius+1.0);
    if(Math.abs(x)>486||Math.abs(z)>486)return;
    if(groundY(x,z)<=water.position.y+.75||blockedPosition(x,z))return;
    const playerDist=Math.hypot(x-player.pos.x,z-player.pos.z);
    const score=nd-playerDist*.08;
    if(score<bestScore){bestScore=score;best={x,z,kind:'cover'};}
  };
  if(typeof cars19!=='undefined')for(const c of cars19){if(!c?.group?.parent||c.wrecked39)continue;consider(c.group.position.x,c.group.position.z,2.5);}
  if(typeof buildingNav50!=='undefined')for(const rec of buildingNav50){if(!rec?.g?.parent||rec.g.visible===false)continue;consider(rec.x,rec.z,Math.max(rec.w||6,rec.d||6)*.52);}
  return best;
}

function flankPoint64(n,s,range){
  const p=n.group.position;
  const ax=p.x-player.pos.x,az=p.z-player.pos.z,al=Math.hypot(ax,az)||1;
  const awayX=ax/al,awayZ=az/al,rightX=-awayZ,rightZ=awayX;
  const side=s.side;
  for(const mult of [1,-1]){
    const x=player.pos.x+awayX*range*.72+rightX*side*mult*rand(18,27);
    const z=player.pos.z+awayZ*range*.72+rightZ*side*mult*rand(18,27);
    if(Math.abs(x)<486&&Math.abs(z)<486&&groundY(x,z)>water.position.y+.75&&!blockedPosition(x,z))return{x,z,kind:'flank'};
  }
  return null;
}

function retreatPoint64(n,range){
  const p=n.group.position,dx=p.x-player.pos.x,dz=p.z-player.pos.z,d=Math.hypot(dx,dz)||1;
  const ux=dx/d,uz=dz/d;
  for(const side of [0,1,-1]){
    const x=clamp(p.x+ux*range+(-uz)*side*8,-484,484),z=clamp(p.z+uz*range+(ux)*side*8,-484,484);
    if(groundY(x,z)>water.position.y+.75&&!blockedPosition(x,z))return{x,z,kind:'retreat'};
  }
  return null;
}

function tacticalThink64(n,s,see,dist,now){
  s.role=weaponRole64(n);
  const pref=preferredRange64(s.role),hp=(n.health??100)/100;

  if(now-lastPlayerShot64.time<1100){
    const hd=Math.hypot(n.group.position.x-lastPlayerShot64.pos.x,n.group.position.z-lastPlayerShot64.pos.z);
    if(hd<lastPlayerShot64.radius){
      n.alert=Math.max(n.alert||0,3.8);s.heardUntil=now+2200;
      if(!see&&hd>5){s.mode='investigate';s.waypoint={x:lastPlayerShot64.pos.x,z:lastPlayerShot64.pos.z,kind:'sound'};}
    }
  }

  if(!see&&(n.alert||0)<1){
    for(const o of npcs){
      if(o===n||o.dead||(o.alert||0)<2)continue;
      if(Math.hypot(o.group.position.x-n.group.position.x,o.group.position.z-n.group.position.z)<18){n.alert=2.2;break;}
    }
  }

  const inCombat=see||(n.alert||0)>0||(n.playerProvoked||0)>0||(s.heardUntil>now);
  if(!inCombat){s.mode='idle';s.waypoint=null;return;}

  if(hp<.34||(s.role==='sniper'&&dist<46)){
    const cover=candidateCover64(n,46);
    s.mode=cover?'cover':'retreat';s.waypoint=cover||retreatPoint64(n,20);s.coverUntil=now+rand(2200,3600);return;
  }

  if((s.role==='shotgun'&&dist>16)||(s.role==='smg'&&dist>31)){
    s.mode='rush';s.waypoint={x:player.pos.x,z:player.pos.z,kind:'rush'};return;
  }

  if((s.role==='sniper'&&dist<pref*.72)||(s.role==='marksman'&&dist<pref*.58)||(s.role==='rifle'&&dist<22)){
    s.mode='retreat';s.waypoint=retreatPoint64(n,16);return;
  }

  const canFlank=dist>18&&dist<70&&s.role!=='shotgun';
  if(canFlank&&Math.random()<(s.role==='sniper'?.18:.34)){
    const fp=flankPoint64(n,s,pref);
    if(fp){s.mode='flank';s.waypoint=fp;s.flankUntil=now+rand(1800,3000);s.side*=-1;return;}
  }

  if((s.role==='rifle'||s.role==='marksman'||s.role==='sniper')&&Math.random()<.26){
    const cover=candidateCover64(n,32);
    if(cover){s.mode='cover';s.waypoint=cover;s.coverUntil=now+rand(1500,2600);return;}
  }

  s.mode='hold';s.waypoint=null;
}

function steerNpc64(n,s,dt){
  if(!s.waypoint)return;
  const p=n.group.position,dx=s.waypoint.x-p.x,dz=s.waypoint.z-p.z,d=Math.hypot(dx,dz);
  if(d<.75){if(s.mode!=='cover')s.waypoint=null;return;}
  const ux=dx/d,uz=dz/d;
  let speed=2.0;
  if(s.mode==='rush')speed=3.45;
  else if(s.mode==='flank')speed=2.75;
  else if(s.mode==='retreat')speed=3.05;
  else if(s.mode==='cover')speed=2.35;
  else if(s.mode==='investigate')speed=2.15;
  const step=Math.min(d,speed*dt),nx=p.x+ux*step,nz=p.z+uz*step;
  if(!blockedPosition(nx,nz)){
    p.x=nx;p.z=nz;p.y=groundY(nx,nz);n.group.rotation.y=Math.atan2(ux,uz)+Math.PI;
    if(n.patrol)n.patrol.set(s.waypoint.x,0,s.waypoint.z);
  }
}
