// Push foliage toward bright readable greens and trunks toward warmer browns.
scene.traverse(o=>{
  if(!o.isMesh||!o.material||!o.material.color)return;
  const c=o.material.color;
  const r=c.r,g=c.g,b=c.b;
  // likely foliage
  if(g>r*1.15&&g>b*.95){
    o.material=o.material.clone();o.material.color.offsetHSL(0,.10,.07);o.material.roughness=Math.min(o.material.roughness??.9,.82);
  }
  // likely dirt/wood/building browns
  else if(r>g*1.08&&g>b*1.05&&r>.18){
    o.material=o.material.clone();o.material.color.offsetHSL(0,.06,.035);
  }
});
