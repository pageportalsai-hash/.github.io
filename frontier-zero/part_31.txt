// Make car collision use the car footprint instead of only its center point.
// This greatly reduces cars clipping their corners through buildings/rocks.
const blockedPositionBeforeCarQA31=blockedPosition;
blockedPosition=function(x,z){
  if(!activeCar19)return blockedPositionBeforeCarQA31(x,z);
  if(blockedPositionBeforeCarQA31(x,z))return true;
  const a=activeCar19.group.rotation.y;
  const forwardX=-Math.sin(a),forwardZ=-Math.cos(a),rightX=Math.cos(a),rightZ=-Math.sin(a);
  const probes=[[1.25,2.15],[-1.25,2.15],[1.25,-2.15],[-1.25,-2.15]];
  for(const [side,front] of probes){
    const px=x+rightX*side+forwardX*front,pz=z+rightZ*side+forwardZ*front;
    if(blockedPositionBeforeCarQA31(px,pz))return true;
  }
  return false;
