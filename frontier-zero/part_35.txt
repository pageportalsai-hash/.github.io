function npcGunFamily35(w){
  const n=(w?.name||'').toLowerCase();
  if(n.includes('shotgun'))return 'shotgun';
  if(n.includes('pistol')||n.includes('revolver')||n.includes('handcannon')||n.includes('cannon'))return 'pistol';
  if(n.includes('smg')||n.includes('machine')||n.includes('compact'))return 'smg';
  if(n.includes('dmr')||n.includes('marksman'))return 'dmr';

}

updateNPCs=function(dt){
  updateNPCsBeforePhaseB35(dt);
  const time=performance.now()*.0065,nearDist=phaseBDesktop35?125:78;
  for(const r of npcRigs35){
    const n=r.n,g=n.group;if(!n||n.dead||!g)continue;
    const dist=Math.hypot(g.position.x-player.pos.x,g.position.z-player.pos.z),near=dist<nearDist;

    const dx=g.position.x-r.prev.x,dz=g.position.z-r.prev.z,speed=Math.hypot(dx,dz)/Math.max(dt,.001);r.prev.copy(g.position);
    const move=clamp(speed/3.5,0,1),run=clamp((speed-2.2)/2.2,0,1),swing=Math.sin(time+r.phase)*(0.58+run*.22)*move;

    r.hit=Math.max(0,(r.hit||0)-dt*4.2);

    r.lShin.rotation.x=Math.max(0,-swing)*.38;r.rShin.rotation.x=Math.max(0,swing)*.38;

  }
