// Replace the Phase-37 collision wrapper with a precomputed version. Buildings are
// static, so recalculating localToWorld for every furniture collider every frame

  if(blockedPositionBeforeInterior37(x,z))return true;
  for(const c of interiorCollision37){
    if(!c.root?.visible||!c.root?.parent?.visible)continue;

const shadowState38=new WeakMap();
function setGroupShadow38(g,on){

  g.traverse(o=>{if(o.isMesh){o.castShadow=on;o.receiveShadow=true;}});

function visibleAncestor38(o){let p=o;while(p){if(p.visible===false)return false;p=p.parent;}return true;}

function updateLOD38(){
  if((lodTick38++%24)!==0)return;
  const px=player.pos.x,pz=player.pos.z;

  // NPCs: near rigs keep dynamic shadows; distant combatants remain fully active
  // but no longer cost dozens of shadow draw calls each.
  for(const n of npcs){
    if(!n?.group||n.dead)continue;const d=Math.hypot(n.group.position.x-px,n.group.position.z-pz);
    setGroupShadow38(n.group,d<(perfDesktop38?(perfTier38===0?62:48):34));
  }
  // Cars: only nearby vehicles cast shadows. Detailed geometry stays visible.
  for(const c of cars19){
    const d=Math.hypot(c.group.position.x-px,c.group.position.z-pz);setGroupShadow38(c.group,d<(perfDesktop38?(perfTier38===0?92:68):48));
  }
  // Authored buildings keep receiving light, but only nearby structures cast.
  for(const g of rebuiltBuildings33){
    const b=g.userData?.bounds33;if(!b)continue;const d=Math.hypot(b.x-px,b.z-pz);
    setGroupShadow38(g,d<(perfDesktop38?(perfTier38===0?115:86):58));
  }

  // Interior distance and actual point lights get progressively tighter under load.
  const interiorRange=perfDesktop38?(perfTier38===0?92:perfTier38===1?70:52):(perfTier38===0?58:44);
  const lightRange=perfTier38===0?32:perfTier38===1?22:0;
  for(const root of interiorRoots37){
    const b=root.userData.bounds37,d=Math.hypot(b.x-px,b.z-pz);root.visible=d<interiorRange;
    if(root.userData.light37)root.userData.light37.visible=lightRange>0&&d<lightRange;
    // Dynamically keep raycast panes active only while their visual interior is active.
    root.traverse(o=>{if(!o.isMesh||!o.userData?.breakable37)return;const br=o.userData.breakable37,idx=blockers.indexOf(o);if(root.visible&&!br.broken&&idx<0)blockers.push(o);else if((!root.visible||br.broken)&&idx>=0)blockers.splice(idx,1);});
  }

  // Decorative exterior groups disappear individually instead of hiding the entire
  // prop layer, avoiding both empty towns and distant draw-call waste.
  for(const g of worldDress37.children){
    const wp=g.getWorldPosition(new THREE.Vector3()),d=Math.hypot(wp.x-px,wp.z-pz);g.visible=d<(perfDesktop38?(perfTier38===0?150:110):82);
  }

  // Low-priority atmospheric extras are the first thing cut at the lowest tier.
  for(const m of mist37)if(perfTier38===2)m.visible=false;
}

let cleanTimer38=0;
function cleanupWorld38(dt){
  cleanTimer38+=dt;if(cleanTimer38<2.2)return;cleanTimer38=0;
  const seen=new Set();
  for(let i=blockers.length-1;i>=0;i--){
    const o=blockers[i];
    if(!o||!o.parent||o.userData?.destroyed37||seen.has(o)){blockers.splice(i,1);continue;}
    if(!visibleAncestor38(o)&&!o.userData?.carHit37){blockers.splice(i,1);continue;}
    seen.add(o);
  }
  if(typeof colliders!=='undefined')for(let i=colliders.length-1;i>=0;i--){const o=colliders[i]?.obj;if(!o||!o.parent||!visibleAncestor38(o))colliders.splice(i,1);}
  if(typeof playerStructures!=='undefined')for(let i=playerStructures.length-1;i>=0;i--){const o=playerStructures[i];if(!o||!o.parent||!visibleAncestor38(o))playerStructures.splice(i,1);}
}

function samplePerf38(now){
  perfFrames38++;
  if(now-perfStart38<4200)return;
  const fps=perfFrames38*1000/(now-perfStart38);perfFrames38=0;perfStart38=now;
  const baseTier=typeof qaRenderTier31!=='undefined'?qaRenderTier31:0;
  if(fps<30&&perfTier38<2){applyPerfTier38(perfTier38+1);perfGood38=0;}
  else if(fps<40&&perfTier38<1){applyPerfTier38(1);perfGood38=0;}
  else if(fps>53&&baseTier<2){perfGood38++;if(perfGood38>=3&&perfTier38>0){applyPerfTier38(perfTier38-1);perfGood38=0;}}
  else perfGood38=0;
}
