function npcLineClear(fromNpc,toNpc){
  if(!fromNpc||!toNpc||fromNpc.dead||toNpc.dead)return false;
  const from=fromNpc.group.position.clone().add(new THREE.Vector3(0,1.75,0));
  const to=toNpc.group.position.clone().add(new THREE.Vector3(0,1.35,0));
  const d=to.clone().sub(from),dist=d.length();if(dist>42)return false;
  raycaster.set(from,d.normalize());raycaster.far=Math.max(.1,dist-.55);
  const hits=raycaster.intersectObjects([terrain,...blockers],true);
  return hits.length===0;
}
function nearestRival(n,maxDist=32){
  let best=null,bestD=maxDist;
  for(const o of npcs){
    if(o===n||o.dead||o.faction===n.faction)continue;
    const d=n.group.position.distanceTo(o.group.position);
    if(d<bestD&&npcLineClear(n,o)){best=o;bestD=d;}
  }
  return best;
}
function npcShootNpc(attacker,target,dist){
  attacker.npcCombatCd=rand(.55,1.0);
  const hitChance=clamp(.58-dist/85,.28,.56);
  tone(attacker.faction==='WARDEN'?92:78,.035,.012,'square');
  if(Math.random()<hitChance){
    target.health-=rand(8,14);target.npcEnemy=attacker;target.alert=Math.max(target.alert,3.5);
    const pt=target.group.position.clone().add(new THREE.Vector3(rand(-.2,.2),rand(.9,1.8),rand(-.2,.2)));impact(pt,0xb75e3b);
    if(target.health<=0)killNpc(target);
  }
}

// Player shooting an NPC explicitly provokes that NPC for ranged retaliation.
const originalShootGunFZ2=shootGun;
shootGun=function(w){
  // Capture NPC health before the original hit-scan resolves, then flag damaged NPCs as provoked.
  const before=new Map();for(const n of npcs)if(!n.dead)before.set(n,n.health);
  originalShootGunFZ2(w);
  for(const n of npcs){
    if(!n.dead&&before.has(n)&&n.health<before.get(n)){n.playerProvoked=8;n.alert=Math.max(n.alert,8);}
  }
};

// Passive NPCs do NOT shoot the player from range. They only auto-engage at ~1 meter.
// Once the player shoots an NPC, that NPC is provoked and may return fire from range.
updateNPCs=function(dt){
  for(const n of npcs){
    if(n.dead)continue;
    const p=n.group.position,toPlayer=player.pos.clone().sub(p);toPlayer.y=0;const playerDist=toPlayer.length();
    n.playerProvoked=Math.max(0,(n.playerProvoked||0)-dt);
    n.npcCombatCd=(n.npcCombatCd||0)-dt;

    // Choose/refresh an opposing NPC target.
    if(!n.npcEnemy||n.npcEnemy.dead||n.npcEnemy.faction===n.faction||n.group.position.distanceTo(n.npcEnemy.group.position)>38){n.npcEnemy=nearestRival(n,30);}
    const rival=n.npcEnemy;
    const rivalDist=rival?n.group.position.distanceTo(rival.group.position):Infinity;
    const rivalVisible=rival?npcLineClear(n,rival):false;

    const passivePlayerAggro=playerDist<=1.15;
    const canRetaliate=n.playerProvoked>0;
    const seePlayer=npcCanSee(n);
    let move=new THREE.Vector3();

    if((passivePlayerAggro||canRetaliate)&&!player.dead){
      const dir=toPlayer.lengthSq()>.001?toPlayer.clone().normalize():new THREE.Vector3(0,0,-1);
      n.alert=Math.max(n.alert,canRetaliate?5:1.5);
      if(canRetaliate&&playerDist>7)move.copy(dir);
      else if(playerDist<1.45)move.set(-dir.z,0,dir.x).multiplyScalar(n.strafe*.55);
      else move.copy(dir);
      n.shotCd-=dt;
      const allowedRange=canRetaliate?34:1.35;
      if(seePlayer&&playerDist<=allowedRange&&n.shotCd<=0){
        n.shotCd=rand(.58,1.08);
        const accuracy=canRetaliate?clamp(.54-playerDist/95,.27,.51):.46;
        if(Math.random()<accuracy)damagePlayer(rand(5,9));else tone(65,.025,.008,'square');
      }
    } else if(rival&&rivalVisible){
      // NPC-vs-NPC firefight.
      const dir=rival.group.position.clone().sub(p);dir.y=0;const nd=dir.length();if(nd>.001)dir.normalize();
      if(rivalDist>13)move.copy(dir);else if(rivalDist<7)move.addScaledVector(dir,-.45);else move.set(-dir.z,0,dir.x).multiplyScalar(n.strafe*.52);
      if(n.npcCombatCd<=0&&rivalDist<30)npcShootNpc(n,rival,rivalDist);
    } else {
      n.alert=Math.max(0,n.alert-dt);
      n.repath-=dt;
      if(n.repath<=0||p.distanceTo(n.patrol)<4){n.repath=rand(4,8);n.patrol.set(p.x+rand(-26,26),0,p.z+rand(-26,26));}
      move.copy(n.patrol).sub(p);move.y=0;if(move.length()>1)move.normalize();
    }

    if(move.lengthSq()>0){
      const fighting=(passivePlayerAggro||canRetaliate||rivalVisible);
      const speed=fighting?3.7:1.5,nx=p.x+move.x*speed*dt,nz=p.z+move.z*speed*dt;
      if(!blockedPosition(nx,nz)){p.x=nx;p.z=nz;}else{n.strafe*=-1;n.patrol.set(p.x+rand(-12,12),0,p.z+rand(-12,12));}
      p.y=groundY(p.x,p.z);n.group.rotation.y=Math.atan2(move.x,move.z)+Math.PI;
    }
  }
};
