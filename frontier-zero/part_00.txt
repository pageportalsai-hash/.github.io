function groundY(x,z){
  const broad = Math.sin(x*0.009)*2.7 + Math.cos(z*0.008)*2.1 + Math.sin((x+z)*0.004)*2.8;
  const detail = Math.sin(x*0.035)*0.55 + Math.cos(z*0.031)*0.45;
  const mountain = Math.max(0, 20 - Math.hypot(x+260,z-250)*0.055);
  return broad + detail + mountain - 1.2;
}
