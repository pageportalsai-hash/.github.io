const raycaster=new THREE.Raycaster();
function cameraBasis(spread=0){
  const dir=new THREE.Vector3(0,0,-1).applyEuler(camera.rotation).normalize();
  if(spread){
    const right=new THREE.Vector3(1,0,0).applyEuler(camera.rotation); const up=new THREE.Vector3(0,1,0).applyEuler(camera.rotation);
    dir.addScaledVector(right,rand(-spread,spread)).addScaledVector(up,rand(-spread,spread)).normalize();
  }
  return dir;
}
function nearestHit(dir, maxDist=130){
  raycaster.set(camera.position,dir); raycaster.far=maxDist;
  const objs=[terrain,...blockers,...npcHitMeshes].filter(o=>o===terrain||o.parent);
  const hits=raycaster.intersectObjects(objs,true);
  return hits[0]||null;
}
function shootGun(w){
  if(reloading) return;
  if(w.mag<=0){tone(70,.035,.025,'square');message('Empty — RELOAD'); return;}
  w.mag--; lastAction=clock.elapsedTime; muzzleLight.intensity=7; setTimeout(()=>muzzleLight.intensity=0,35); gunHolder.position.z=-.67; setTimeout(()=>gunHolder.position.z=-.72,55);
  tone(w.name==='Shotgun'?85:w.name==='Pistol'?145:115,.045,w.name==='Shotgun'?.075:.045,'sawtooth');
  for(let p=0;p<w.pellets;p++){
    const hit=nearestHit(cameraBasis(w.spread), w.name==='Shotgun'?80:160); if(!hit)continue;
    const obj=hit.object;
    let npc=obj.userData.npc; if(!npc && obj.parent) npc=obj.parent.userData.npc;
    if(npc && !npc.dead){ npc.health-=w.damage; npc.alert=5; impact(hit.point,0xb75e3b); if(npc.health<=0) killNpc(npc); continue; }
    const res=findResource(obj); if(res && !res.dead){ impact(hit.point,0xc7b291); }
    else impact(hit.point,0xb7c0c0);
  }
  updateHUD();
}
function findResource(obj){ let o=obj; for(let i=0;i<3&&o;i++,o=o.parent){ if(o.userData?.resource)return o.userData.resource; } return null; }
