function blockedPosition(x,z){
  for(const c of colliders){
    if(!c.obj.parent) continue;
    const p=new THREE.Vector3(); c.obj.getWorldPosition(p);
    if(Math.hypot(x-p.x,z-p.z)<c.radius+0.55) return true;
  }
  for(const s of playerStructures){
    if(!s.parent) continue;
    const box=new THREE.Box3().setFromObject(s).expandByScalar(.35);
    if(x>box.min.x && x<box.max.x && z>box.min.z && z<box.max.z) return true;
  }
  return false;
}

function updateJoy(e){
  const r=hud.joyBase.getBoundingClientRect(), cx=r.left+r.width/2, cy=r.top+r.height/2;
  let dx=e.clientX-cx, dy=e.clientY-cy; const max=42, len=Math.hypot(dx,dy)||1, k=Math.min(1,max/len); dx*=k;dy*=k;
  joy.x=dx/max; joy.y=dy/max; hud.joyKnob.style.transform=`translate(${dx}px,${dy}px)`;
}
