function miniWorldToPx(x,z){return {x:(x+500)/1000*276,y:(z+500)/1000*276};}
function drawMiniMap(){
  const c=miniCtx;c.clearRect(0,0,276,276);
  const g=c.createLinearGradient(0,0,276,276);g.addColorStop(0,'#31443b');g.addColorStop(.5,'#46543d');g.addColorStop(1,'#4e3d31');c.fillStyle=g;c.fillRect(0,0,276,276);
  c.strokeStyle='rgba(255,255,255,.15)';c.lineWidth=2;
  for(let i=1;i<4;i++){const p=i*69;c.beginPath();c.moveTo(p,0);c.lineTo(p,276);c.stroke();c.beginPath();c.moveTo(0,p);c.lineTo(276,p);c.stroke();}
  // towns / cities from the current world layout
  if(typeof settlementLayout!=='undefined'){
    for(const s of settlementLayout){if(Math.abs(s.x)>500||Math.abs(s.z)>500)continue;const p=miniWorldToPx(s.x,s.z);c.fillStyle=s.type==='city'?'#e7d8a7':'#c5b98b';c.fillRect(p.x-3,p.y-3,6,6);}
  }
  // NPCs: red dots, living only
  c.fillStyle='#ef6258';
  for(const n of npcs){if(n.dead||!n.group.visible)continue;const p=miniWorldToPx(n.group.position.x,n.group.position.z);c.beginPath();c.arc(p.x,p.y,2.2,0,Math.PI*2);c.fill();}
  // player arrow
  const p=miniWorldToPx(player.pos.x,player.pos.z);c.save();c.translate(p.x,p.y);c.rotate(-player.yaw);c.fillStyle='#ffffff';c.beginPath();c.moveTo(0,-8);c.lineTo(5,6);c.lineTo(0,3);c.lineTo(-5,6);c.closePath();c.fill();c.restore();
  c.strokeStyle='rgba(255,255,255,.55)';c.lineWidth=2;c.strokeRect(1,1,274,274);
  c.fillStyle='rgba(0,0,0,.60)';c.fillRect(6,6,58,20);c.fillStyle='#fff';c.font='bold 13px system-ui';c.fillText('MAP',14,21);
}

let lastWeaponLook13='';
function applyWeaponLook13(){
  const w=currentGear();if(!w||w.type!=='gun')return;
  const key=(w.rarity||'Common')+'|'+w.name;if(key===lastWeaponLook13)return;lastWeaponLook13=key;
  const body=baseGunParts[0],stock=baseGunParts[1],barrel=baseGunParts[2];if(!body||!stock||!barrel)return;
  body.scale.set(1,1,1);stock.scale.set(1,1,1);barrel.scale.set(1,1,1);body.position.set(0,0,-.15);stock.position.set(0,-.015,.28);barrel.position.set(0,.015,-.7);
  scope13.visible=false;foregrip13.visible=false;suppressor13.visible=false;
  const name=w.name.toLowerCase();
  if(name.includes('smg')){body.scale.set(.88,.82,.72);stock.scale.set(.72,.72,.58);barrel.scale.set(.82,.82,.52);body.position.z=-.05;barrel.position.z=-.48;foregrip13.visible=true;}
  else if(name.includes('shotgun')){body.scale.set(1.08,1.02,1.15);stock.scale.set(1.08,1.02,1.15);barrel.scale.set(1.28,1.28,1.42);barrel.position.z=-.85;}
  else if(name.includes('pistol')||name.includes('handcannon')||name.includes('cannon')){body.scale.set(.68,.82,.54);stock.scale.set(.32,.75,.30);barrel.scale.set(name.includes('cannon')?1.28:.86,name.includes('cannon')?1.28:.86,.52);body.position.z=-.02;stock.position.z=.12;barrel.position.z=-.38;}
  else if(name.includes('carbine')){body.scale.set(.95,.92,.82);stock.scale.set(.82,.88,.76);barrel.scale.set(.92,.92,.75);foregrip13.visible=true;}
  else {scope13.visible=(w.rarity==='Rare'||w.rarity==='Epic'||w.rarity==='Legendary'||w.rarity==='Mythic'||w.rarity==='Secret');foregrip13.visible=(w.rarity==='Epic'||w.rarity==='Legendary'||w.rarity==='Mythic'||w.rarity==='Secret');}

}
