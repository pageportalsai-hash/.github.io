// Procedural asphalt texture gives roads surface variation without downloading external assets.
function makeAsphaltTexture(){
  const c=document.createElement('canvas');c.width=c.height=128;const ctx=c.getContext('2d');
  ctx.fillStyle='#343638';ctx.fillRect(0,0,128,128);
  for(let i=0;i<1600;i++){const v=(35+Math.random()*45)|0;ctx.fillStyle=`rgba(${v},${v},${v},${rand(.04,.16)})`;const s=Math.random()<.94?1:2;ctx.fillRect(Math.random()*128,Math.random()*128,s,s);}
  for(let i=0;i<9;i++){ctx.strokeStyle='rgba(20,20,20,.18)';ctx.lineWidth=rand(.4,1.2);ctx.beginPath();let x=rand(0,128),y=rand(0,128);ctx.moveTo(x,y);for(let k=0;k<4;k++){x+=rand(-15,15);y+=rand(8,24);ctx.lineTo(x,y);}ctx.stroke();}
  const t=new THREE.CanvasTexture(c);t.wrapS=t.wrapT=THREE.RepeatWrapping;t.repeat.set(4,4);t.colorSpace=THREE.SRGBColorSpace;return t;
}
