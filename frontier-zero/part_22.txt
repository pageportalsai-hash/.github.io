// Rebuilds the existing drivable cars with more realistic multi-part models
// while preserving their positions, controls, physics and ~50 mph speed cap.

const detailedCarWheels22=[];
function carMesh22(parent,geo,mat,x,y,z,rx=0,ry=0,rz=0){
  const m=new THREE.Mesh(geo,mat);m.position.set(x,y,z);m.rotation.set(rx,ry,rz);m.castShadow=true;m.receiveShadow=true;parent.add(m);return m;
}
function rebuildCar22(car,i){
  const g=car.group;
  // Remove the old placeholder body/cabin/wheels but keep the same car group.
  while(g.children.length)g.remove(g.children[g.children.length-1]);

  const baseColor=carColors19[i%carColors19.length];
  const paint=new THREE.MeshPhysicalMaterial({color:baseColor,roughness:.25,metalness:.62,clearcoat:.72,clearcoatRoughness:.18});
  const paintDark=new THREE.MeshPhysicalMaterial({color:new THREE.Color(baseColor).multiplyScalar(.68),roughness:.3,metalness:.58,clearcoat:.55});
  const black=new THREE.MeshStandardMaterial({color:0x121416,roughness:.68,metalness:.18});
  const tire=new THREE.MeshStandardMaterial({color:0x101112,roughness:.94,metalness:.03});
  const rim=new THREE.MeshStandardMaterial({color:0xaeb6bb,roughness:.24,metalness:.88});
  const glass=new THREE.MeshPhysicalMaterial({color:0x172c38,roughness:.08,metalness:.05,transmission:.14,transparent:true,opacity:.72});
  const chrome=new THREE.MeshStandardMaterial({color:0xbfc6ca,roughness:.16,metalness:.92});
  const headlight=new THREE.MeshStandardMaterial({color:0xfff2c8,emissive:0xffd98b,emissiveIntensity:.65,roughness:.18,metalness:.05});
  const tail=new THREE.MeshStandardMaterial({color:0x9b1717,emissive:0x5c0707,emissiveIntensity:.55,roughness:.25});
  const amber=new THREE.MeshStandardMaterial({color:0xf3a51f,emissive:0x9a4f00,emissiveIntensity:.35,roughness:.28});

  // Main shell: lower chassis, sculpted mid body, hood and trunk.
  carMesh22(g,new THREE.BoxGeometry(3.35,.48,5.55),paint,0,.58,0);
  const mid=carMesh22(g,new THREE.BoxGeometry(3.20,.58,4.65),paint,0,.92,.05);mid.scale.set(1,.95,1);
  const hood=carMesh22(g,new THREE.BoxGeometry(3.08,.34,1.55),paint,0,1.20,-1.78,-.04,0,0);
  const trunk=carMesh22(g,new THREE.BoxGeometry(3.05,.31,1.18),paint,0,1.16,2.02,.035,0,0);

  // Cabin and pillars give the silhouette actual shape instead of one glass box.
  const roof=carMesh22(g,new THREE.BoxGeometry(2.55,.16,2.15),paintDark,0,2.05,.18);
  const windshield=carMesh22(g,new THREE.BoxGeometry(2.46,.78,.08),glass,0,1.68,-.86,-.48,0,0);
  const rearGlass=carMesh22(g,new THREE.BoxGeometry(2.42,.72,.08),glass,0,1.69,1.22,.46,0,0);
  const leftGlass=carMesh22(g,new THREE.BoxGeometry(.07,.70,1.62),glass,-1.28,1.66,.18,0,0,0);
  const rightGlass=carMesh22(g,new THREE.BoxGeometry(.07,.70,1.62),glass,1.28,1.66,.18,0,0,0);
  carMesh22(g,new THREE.BoxGeometry(.15,.77,.16),paintDark,-1.29,1.68,-.10);
  carMesh22(g,new THREE.BoxGeometry(.15,.77,.16),paintDark,1.29,1.68,-.10);

  // Bumpers, grille, skirts and handles.
  carMesh22(g,new THREE.BoxGeometry(3.10,.19,.18),black,0,.62,-2.82);
  carMesh22(g,new THREE.BoxGeometry(3.06,.18,.16),black,0,.60,2.82);
  carMesh22(g,new THREE.BoxGeometry(1.62,.31,.09),black,0,.83,-2.89);
  for(const x of [-.87,-.45,0,.45,.87])carMesh22(g,new THREE.BoxGeometry(.06,.25,.025),chrome,x,.83,-2.95);
  carMesh22(g,new THREE.BoxGeometry(.13,.12,3.30),paintDark,-1.68,.66,.15);
  carMesh22(g,new THREE.BoxGeometry(.13,.12,3.30),paintDark,1.68,.66,.15);
  for(const x of [-1.43,1.43])for(const z of [-.25,.78])carMesh22(g,new THREE.BoxGeometry(.18,.045,.35),chrome,x,1.20,z);

  // Mirrors.
  for(const x of [-1.67,1.67]){
    const mirror=carMesh22(g,new THREE.BoxGeometry(.24,.18,.42),paintDark,x,1.53,-.58,0,0,x<0?-.08:.08);
    mirror.scale.set(1,1,.78);
  }

  // Headlights, tail lights, indicators and license plate.
  for(const x of [-1.05,1.05]){
    carMesh22(g,new THREE.BoxGeometry(.74,.24,.08),headlight,x,1.03,-2.83,-.03,0,0);
    carMesh22(g,new THREE.BoxGeometry(.58,.24,.08),tail,x,1.02,2.84,.02,0,0);
    carMesh22(g,new THREE.BoxGeometry(.18,.18,.09),amber,x<0?-1.49:1.49,1.00,-2.84);
  }
  carMesh22(g,new THREE.BoxGeometry(.72,.24,.045),new THREE.MeshStandardMaterial({color:0xe8e8dc,roughness:.5}),0,.64,2.91);

  // Detailed wheels with tires, rims and center hubs.
  car.wheels22=[];
  for(const sx of [-1.68,1.68])for(const sz of [-1.78,1.78]){
    const wheel=new THREE.Group();wheel.position.set(sx,.55,sz);g.add(wheel);
    const t=carMesh22(wheel,new THREE.CylinderGeometry(.52,.52,.38,18),tire,0,0,0,0,0,Math.PI/2);
    const r=carMesh22(wheel,new THREE.CylinderGeometry(.31,.31,.40,12),rim,0,0,0,0,0,Math.PI/2);
    carMesh22(wheel,new THREE.CylinderGeometry(.11,.11,.42,12),black,0,0,0,0,0,Math.PI/2);
    for(let s=0;s<6;s++){
      const a=s/6*Math.PI*2;
      const spoke=carMesh22(wheel,new THREE.BoxGeometry(.035,.035,.25),chrome,0,Math.cos(a)*.16,Math.sin(a)*.16,a,0,0);
      spoke.rotation.x=a;
    }
    car.wheels22.push(wheel);detailedCarWheels22.push(wheel);
  }

  // Subtle interior: seats and dashboard visible through the windows.
  const interior=new THREE.MeshStandardMaterial({color:0x252321,roughness:.88});
  for(const x of [-.63,.63])for(const z of [-.20,.78]){
    carMesh22(g,new THREE.BoxGeometry(.52,.72,.48),interior,x,1.18,z);
    carMesh22(g,new THREE.BoxGeometry(.54,.58,.18),interior,x,1.48,z+.13,-.15,0,0);
  }
  carMesh22(g,new THREE.BoxGeometry(2.25,.18,.54),black,0,1.34,-.58,-.10,0,0);
  carMesh22(g,new THREE.TorusGeometry(.22,.035,8,16),black,-.62,1.47,-.84,Math.PI/2,0,0);

  // Small exhaust tips.
  carMesh22(g,new THREE.CylinderGeometry(.075,.075,.28,10),chrome,-.92,.42,2.90,Math.PI/2,0,0);
  carMesh22(g,new THREE.CylinderGeometry(.075,.075,.28,10),chrome,.92,.42,2.90,Math.PI/2,0,0);
}
for(let i=0;i<cars19.length;i++)rebuildCar22(cars19[i],i);

// Animate wheel rotation based on actual vehicle speed.
const updatePlayerBeforeCarVisual22=updatePlayer;
updatePlayer=function(dt){
  updatePlayerBeforeCarVisual22(dt);
  for(const c of cars19){
    if(!c.wheels22)continue;
    const spin=(c.speed||0)*dt/.52;
    for(const w of c.wheels22)w.rotation.x-=spin;
  }
};

message('Cars visually upgraded · detailed bodywork · lights · rims · mirrors · interiors',3.2);
